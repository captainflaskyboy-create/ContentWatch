cmake_minimum_required(VERSION 3.20)
project(ContentWatchSpout LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Spout2 is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SPOUT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/spout-sdk" CACHE PATH "Pinned Spout2 checkout")
set(GL "${SPOUT_SOURCE}/SPOUTSDK/SpoutGL")
set(DX "${SPOUT_SOURCE}/SPOUTSDK/SpoutDirectX/SpoutDX")
add_library(SpoutSdk STATIC "${DX}/SpoutDX.cpp" "${GL}/SpoutCopy.cpp" "${GL}/SpoutDirectX.cpp" "${GL}/SpoutFrameCount.cpp" "${GL}/SpoutSenderNames.cpp" "${GL}/SpoutSharedMemory.cpp" "${GL}/SpoutUtils.cpp")
target_compile_definitions(SpoutSdk PUBLIC SPOUT_BUILD_STATIC)
target_include_directories(SpoutSdk PUBLIC "${DX}" "${GL}")
target_link_libraries(SpoutSdk PUBLIC d3d11 dxgi dxguid opengl32 winmm version shlwapi psapi shell32 user32 gdi32 ole32 advapi32 comdlg32)
add_library(ContentWatch.Spout SHARED bridge.cpp)
target_link_libraries(ContentWatch.Spout PRIVATE SpoutSdk)
add_executable(SpoutSmoke smoke.cpp)
target_link_libraries(SpoutSmoke PRIVATE SpoutSdk ContentWatch.Spout)
add_executable(SpoutInspect inspect.cpp)
target_link_libraries(SpoutInspect PRIVATE SpoutSdk)
foreach(target ContentWatch.Spout SpoutSmoke SpoutInspect)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../artifacts/native")
endforeach()
